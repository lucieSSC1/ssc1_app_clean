CREATE TABLE ecrit (
    id        SERIAL PRIMARY KEY,

    -- Chronologie
    glob_id   INTEGER,        -- Lien vers global_info

    titre     TEXT,
    type      TEXT,           -- Poème, texte, roman, etc.
    ecrit_url TEXT,
    notes     TEXT,

    CONSTRAINT fk_ecrit_glob_info
        FOREIGN KEY (glob_id)
        REFERENCES glob_info(id)
        ON DELETE SET NULL
);
